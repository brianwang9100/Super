CREATE TABLE "bibleAnnotation" ("id" TEXT PRIMARY KEY NOT NULL, "target" TEXT NOT NULL, "bookId" TEXT NOT NULL, "chapterNumber" INTEGER, "verseStart" INTEGER, "verseEnd" INTEGER, "summary" TEXT NOT NULL, "source" TEXT NOT NULL, "modelId" TEXT NOT NULL, "createdAt" DATETIME NOT NULL);
CREATE INDEX "bibleAnnotation_on_bookId_chapterNumber_verseEnd" ON "bibleAnnotation"("bookId", "chapterNumber", "verseEnd");
CREATE INDEX "bibleAnnotation_on_target_bookId" ON "bibleAnnotation"("target", "bookId");
CREATE TABLE "bibleBookmark" ("id" TEXT PRIMARY KEY NOT NULL, "colorId" TEXT NOT NULL, "bookId" TEXT NOT NULL, "chapterNumber" INTEGER NOT NULL, "createdAt" DATETIME NOT NULL);
CREATE UNIQUE INDEX "bibleBookmark_on_bookId_chapterNumber" ON "bibleBookmark"("bookId", "chapterNumber");
CREATE UNIQUE INDEX "bibleBookmark_on_colorId" ON "bibleBookmark"("colorId");
CREATE TABLE "bibleHighlight" ("id" TEXT PRIMARY KEY NOT NULL, "bookId" TEXT NOT NULL, "chapterNumber" INTEGER NOT NULL, "verseNumber" INTEGER NOT NULL, "colorId" TEXT NOT NULL, "createdAt" DATETIME NOT NULL, "updatedAt" DATETIME NOT NULL, "deletedAt" DATETIME);
CREATE UNIQUE INDEX "bibleHighlight_on_bookId_chapterNumber_verseNumber" ON "bibleHighlight"("bookId", "chapterNumber", "verseNumber");
CREATE INDEX "bibleHighlight_on_deletedAt" ON "bibleHighlight"("deletedAt");
CREATE TABLE "bibleNote" ("id" TEXT PRIMARY KEY NOT NULL, "target" TEXT NOT NULL, "bookId" TEXT NOT NULL, "chapterNumber" INTEGER, "verseStart" INTEGER, "verseEnd" INTEGER, "body" TEXT NOT NULL, "source" TEXT NOT NULL, "modelId" TEXT, "createdAt" DATETIME NOT NULL, "updatedAt" DATETIME NOT NULL);
CREATE INDEX "bibleNote_on_bookId_chapterNumber_verseEnd" ON "bibleNote"("bookId", "chapterNumber", "verseEnd");
CREATE INDEX "bibleNote_on_target_bookId" ON "bibleNote"("target", "bookId");
CREATE TABLE "bibleReadingPosition" ("id" TEXT PRIMARY KEY NOT NULL, "bookId" TEXT NOT NULL, "chapterNumber" INTEGER NOT NULL, "translationId" TEXT NOT NULL, "updatedAt" DATETIME NOT NULL);
CREATE TABLE "bulkAnnotationRun" ("id" TEXT PRIMARY KEY NOT NULL, "status" TEXT NOT NULL, "modelId" TEXT NOT NULL, "haltReason" TEXT, "createdAt" DATETIME NOT NULL, "updatedAt" DATETIME NOT NULL, "completedAt" DATETIME, "overwriteExisting" BOOLEAN NOT NULL DEFAULT 0, CHECK ((status IN ('completed', 'failed', 'cancelled')) = (completedAt IS NOT NULL)));
CREATE INDEX "bulkAnnotationRun_on_completedAt" ON "bulkAnnotationRun"("completedAt");
CREATE INDEX "bulkAnnotationRun_on_status" ON "bulkAnnotationRun"("status");
CREATE TABLE "bulkAnnotationRunUnit" ("id" TEXT PRIMARY KEY NOT NULL, "runId" TEXT NOT NULL REFERENCES "bulkAnnotationRun"("id") ON DELETE CASCADE, "ordinal" INTEGER NOT NULL, "kind" TEXT NOT NULL, "bookId" TEXT NOT NULL, "bookName" TEXT NOT NULL, "chapterNumber" INTEGER, "state" TEXT NOT NULL, "attemptCount" INTEGER NOT NULL DEFAULT 0, "producedCount" INTEGER NOT NULL DEFAULT 0, "errorMessage" TEXT, "updatedAt" DATETIME NOT NULL);
CREATE INDEX "bulkAnnotationRunUnit_on_runId_ordinal" ON "bulkAnnotationRunUnit"("runId", "ordinal");
CREATE TABLE narrationSettings (
    id TEXT PRIMARY KEY NOT NULL,
    scope TEXT NOT NULL UNIQUE CHECK (scope = 'narration'),
    enabled BOOLEAN,
    sourceId TEXT, sourceName TEXT, keyRef TEXT,
    ownsKey BOOLEAN NOT NULL,
    preferredVoiceId TEXT, lastAppleVoiceId TEXT,
    rate DOUBLE NOT NULL,
    revision INTEGER NOT NULL,
    retiredKeyRefs TEXT NOT NULL,
    updatedAt DATETIME NOT NULL
);
